mostar_negotiations_effect = {
	clr_global_flag = montenegro_ottomans_neogotiations
	MNT = {  }
	OTO = {  }
	if = {
		limit = {
			MNT = {
				OR = {
					has_country_flag = have_indenpendence_montengro
					has_country_flag = not_gained_independence
				}
				OR = {
					has_country_flag = have_outer_montenegro
					has_country_flag = not_gained_outer_montenegro
				}
			}
		}
		MNT = {
			clr_country_flag = MNT_agreed
			clr_country_flag = OTO_agreed
			set_country_flag = MNT_mostar_resolution_pending
			country_event = OTO_battleofgrahovac.24
			country_event = { id = OTO_battleofgrahovac.16 days = 20 }
		}
		OTO = { country_event = OTO_battleofgrahovac.24 }
	}
	else = {
		MNT = { country_event = OTO_battleofgrahovac.23 }
		OTO = { country_event = OTO_battleofgrahovac.23 }
	}
}

add_prestige_battle_of_grahovac = {
	custom_effect_tooltip = add_negotiations_effect
	set_temp_variable = {
		var = add_negotiations_bonus_MNT
		value = 0
	}
	if = {
		limit = { has_country_flag = not_gained_independence }
		add_to_temp_variable = { add_negotiations_bonus = -1 }
	}
	if = {
		limit = { has_country_flag = have_indenpendence_montengro }
		add_to_temp_variable = { add_negotiations_bonus = 3 }
	}
	if = {
		limit = { has_country_flag = have_outer_montenegro }
		add_to_temp_variable = { add_negotiations_bonus = 2 }
	}
	if = {
		limit = { has_country_flag = not_gained_outer_montenegro }
		add_to_temp_variable = { add_negotiations_bonus = -1 }
	}
	if = {
		limit = { has_country_flag = have_arms_import }
		add_to_temp_variable = { add_negotiations_bonus = 1 }
	}
	set_temp_variable = { eoanb_prestige_change = add_negotiations_bonus }
	eoanb_change_prestige_score = yes
	#clr_global_flag = have_indenpendence_montengro
	#clr_global_flag = not_gained_independence
	#clr_global_flag = not_gained_outer_montenegro
	#have_outer_montenegro deliberately omitted , if you remove the flag the evaluation of the war of 1862 will not work
	#clr_global_flag = MNT_neogotiate_arms_import
	#clr_global_flag = deport_mirko_petrovic_njegos
	#clr_global_flag = protect_muslim_landlords
	#clr_global_flag = demanded_debt_repayment
}

MNT_start_war_1858 = {
	# MNT AUSTRIAN WAR COST
	hidden_effect = { MNT_crisis_austrian_war_cost = yes }
	MNT = { if = { limit = { has_country_flag = MNT_crisis_initialized has_country_flag = MNT_crisis_active } set_variable = { MNT_crisis_phase = 3 } clr_country_flag = MNT_crisis_reply_pending if = { limit = { NOT = { has_completed_focus = MNT_focus_defend_the_nation_n } } complete_national_focus = MNT_focus_defend_the_nation_n } } }
	if = {
		limit = { NOT = { has_global_flag = MNT_war_1858 } }
		clr_global_flag = montenegro_ottomans_neogotiations
		clr_global_flag = OTO_herzegovina_uprising_1857
		OTO = {

			set_variable = { available_lives = 4 }
			set_variable = { OTO_max_limit = 1 }
		}
		MNT = {

			clr_country_flag = MNT_mostar_resolution_pending
			set_variable = { available_lives = 4 }
		}
		if = {
			limit = { MNT = { is_subject_of = OTO } }
			OTO = { end_puppet = MNT }
		}
		set_global_flag = MNT_war_1858
		if = { limit = { NOT = { MNT = { has_country_flag = MNT_crisis_initialized } } }
			OTO = { activate_mission = MNT_OTO_mission_war_1858 }
			MNT = { activate_mission = MNT_OTO_mission_war_1858 }
		}
	}
}

		clamp_variable = { var = MNT_crisis_vukalovic_trust min = 0 max = 100 }
		clamp_variable = {
			var = MNT_crisis_strength min = 0 max = 100
		}
		clamp_variable = {
			var = MNT_crisis_supplies min = 0 max = 90
		}
		clamp_variable = {
			var = MNT_crisis_escalation min = 0 max = 100
		}
		clamp_variable = {
			var = MNT_crisis_reputation min = 0 max = 100
		}
		clamp_variable = {
			var = MNT_crisis_france min = 0 max = 3
		}
		clamp_variable = {
			var = MNT_crisis_russia min = 0 max = 3
		}
		clamp_variable = {
			var = MNT_crisis_austria min = 0 max = 3
		}
		set_variable = {
			MNT_crisis_strength_bar = MNT_crisis_strength
		}
		set_variable = {
			MNT_crisis_supply_bar = MNT_crisis_supplies
		}
		divide_variable = {
			MNT_crisis_supply_bar = 0.9
		}
		set_variable = {
			MNT_crisis_escalation_bar = MNT_crisis_escalation
		}
		# Whole points: one per 20 strength, one per 25 credibility, two per patron level.
		set_variable = {
			MNT_crisis_leverage = MNT_crisis_strength
		}
		divide_variable = {
			MNT_crisis_leverage = 20
		}
		round_variable = MNT_crisis_leverage set_temp_variable = {
			MNT_crisis_credit = MNT_crisis_reputation
		}
		divide_temp_variable = {
			MNT_crisis_credit = 25
		}
		round_temp_variable = MNT_crisis_credit add_to_variable = {
			MNT_crisis_leverage = MNT_crisis_credit
		}
		set_temp_variable = {
			MNT_crisis_patrons = MNT_crisis_france
		}
		add_to_temp_variable = {

MNT_crisis_week = {
	if = {
		limit = {
			MNT_crisis_covert_phase = yes has_completed_focus = MNT_focus_guerrilla_war
		}
		add_to_variable = {
			MNT_crisis_days = 7
		}
		if = {
			limit = {
				check_variable = {
					MNT_crisis_supplies > 6
				}
			}
			subtract_from_variable = {
				MNT_crisis_supplies = 7
			}
			add_to_variable = {
				MNT_crisis_strength = 1
			}
		}
		else = {
			subtract_from_variable = {
				MNT_crisis_strength = 6
			}
			set_variable = {
				MNT_crisis_supplies = 0
			}
		}
		add_to_variable = {
			MNT_crisis_escalation = 1
		}
		if = {
			limit = {
				OTO = {
					num_divisions_in_states = {
						count > 4 states = {
							1205
						}
					}
				}
			}
			subtract_from_variable = {
				MNT_crisis_strength = 2
			}
		}
		MNT_crisis_refresh = yes if = {
			limit = {
				check_variable = {
					MNT_crisis_days > 83
				}
				NOT = {
					has_country_flag = MNT_crisis_incident_seen
				}
			}
			set_country_flag = MNT_crisis_incident_seen country_event = {
				id = MNT_crisis.4 days = 1
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = {
						MNT_crisis_days > 195
					}
					check_variable = {
						MNT_crisis_strength < 11
					}
					check_variable = {
						MNT_crisis_escalation > 84
					}
				}
			}
			MNT_crisis_begin_talks = yes
		}
	}
	else_if = {
		limit = {
			MNT_crisis_talks_phase = yes
		}
		add_to_variable = {
			MNT_crisis_phase_days = 7
		}
		if = {
			limit = {
				check_variable = {
					MNT_crisis_phase_days > 55
				}
				NOT = {
					has_country_flag = MNT_crisis_deadline_seen
				}
			}
			set_country_flag = MNT_crisis_deadline_seen country_event = {
				id = MNT_crisis.14 days = 1
			}
		}
	}
	MNT_crisis_refresh = yes
}

MNT_frontier_start = {
	if = {
		limit = { NOT = { has_country_flag = MNT_frontier_running } NOT = { has_country_flag = MNT_frontier_war_record } has_war_with = OTO }
		MNT_frontier_initialize = yes
		MNT_frontier_remove_old_missions = yes
		set_country_flag = MNT_frontier_running
		set_variable = { available_lives = 4 }
		OTO = { set_variable = { available_lives = 4 } }
		set_variable = { MNT_frontier_days = 0 }
		set_variable = { MNT_frontier_pressure = 10 }
		set_temp_variable = { MNT_frontier_backing = MNT_crisis_france }
		add_to_temp_variable = { MNT_frontier_backing = MNT_crisis_russia }
		add_to_temp_variable = { MNT_frontier_backing = MNT_crisis_austria }
		multiply_temp_variable = { MNT_frontier_backing = 2 }
		add_to_variable = { MNT_frontier_pressure = MNT_frontier_backing }
		OTO = {
			if = { limit = { NOT = { has_template = "Herzegovina Frontier Detachment" } }
				division_template = { name = "Herzegovina Frontier Detachment" is_locked = yes
					regiments = { irregular_infantry = { x = 0 y = 0 } irregular_infantry = { x = 0 y = 1 } }
				}
			}
		}
		MNT_frontier_update_cordon = yes
		if = { limit = { NOT = { has_country_flag = MNT_frontier_cordon_suspended } }
			1204 = { MNT_frontier_evacuate_state = yes }
			1205 = { MNT_frontier_evacuate_state = yes }
			1202 = { MNT_frontier_evacuate_state = yes }
			225 = { MNT_frontier_evacuate_state = yes }
			966 = { MNT_frontier_evacuate_state = yes }
			44 = { MNT_frontier_evacuate_state = yes }
			104 = { MNT_frontier_evacuate_state = yes }
			1208 = { MNT_frontier_evacuate_state = yes }
			1272 = { MNT_frontier_evacuate_state = yes }
		}
		# The small, bounded loop creates exactly the committed prewar garrison.
		for_loop_effect = { start = 1 end = OTO.MNT_frontier_garrison compare = less_than_or_equals value = MNT_frontier_spawn_index MNT_frontier_spawn_one = yes }
		MNT_frontier_update_cordon = yes
		MNT_frontier_refresh = yes
		# MNT DIPLOMACY WAR HOOK
		if = { limit = { has_country_flag = MNT_diplomacy_initialized } MNT_diplomacy_war_start = yes }
		country_event = { id = MNT_frontier.1 days = 1 }
		OTO = { country_event = { id = MNT_frontier.1 days = 1 } }
	}
}
